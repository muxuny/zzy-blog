# 后台仪表盘科技控制台 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 将后台仪表盘升级为“内容运营 + 阅读反馈 + 资源健康”的科技控制台，并让所有新增统计由后端 overview 聚合接口提供。

**Architecture:** 扩展现有 `GET /api/admin/dashboard/overview`，在 `AdminDashboardOverview` 中新增流量、阅读、收藏和资源四组 summary。聚合查询使用现有表，不新增业务表或业务字段；新增的 SQL 全部显式过滤 `deleted = 0`。前端继续使用 `buildDashboardStats` 归一化数据，并按已确认原型重做 `Dashboard.vue`。

**Tech Stack:** Vue 3 `<script setup>`、Element Plus、Node.js `node:test` 源码契约测试、Spring Boot、MyBatis-Plus、JUnit 5 + AssertJ、MySQL 增量索引脚本。

```xml
<select id="selectRecentReadingCount" resultType="long">
    SELECT COUNT(1)
    FROM article_reading_history h
    WHERE h.deleted = 0
      AND h.last_read_at &gt;= #{start}
      AND h.last_read_at &lt;= #{end}
</select>

<select id="selectActiveReaderCount" resultType="long">
    SELECT COUNT(DISTINCT h.user_id)
    FROM article_reading_history h
    WHERE h.deleted = 0
      AND h.last_read_at &gt;= #{start}
</select>

<select id="selectAverageProgress" resultType="long">
    SELECT COALESCE(ROUND(AVG(h.progress_percent)), 0)
    FROM article_reading_history h
    WHERE h.deleted = 0
      AND h.last_read_at &gt;= #{start}
</select>

<select id="selectReadingDateBuckets" resultType="com.blog.dto.DashboardDateBucket">
    SELECT DATE_FORMAT(h.last_read_at, '%Y-%m-%d') AS date,
           COUNT(1) AS count
    FROM article_reading_history h
    WHERE h.deleted = 0
      AND h.last_read_at &gt;= #{start}
      AND h.last_read_at &lt; #{end}
    GROUP BY DATE_FORMAT(h.last_read_at, '%Y-%m-%d')
    ORDER BY date ASC
</select>

<select id="selectReadingProgressBuckets" resultType="com.blog.dto.DashboardProgressBucket">
    SELECT CASE
        WHEN h.progress_percent &gt;= 90 THEN '已读完'
        WHEN h.progress_percent &gt;= 60 THEN '接近读完'
        WHEN h.progress_percent &gt;= 25 THEN '阅读中'
        ELSE '浅读'
    END AS label,
    COUNT(1) AS count
    FROM article_reading_history h
    WHERE h.deleted = 0
      AND h.last_read_at &gt;= #{start}
    GROUP BY label
</select>
```

    SELECT f.article_id AS articleId,
           COALESCE(a.title, f.title_snapshot) AS title,
           COUNT(1) AS favoriteCount
    FROM article_favorite f
    LEFT JOIN article a ON a.id = f.article_id
    WHERE f.deleted = 0
    GROUP BY f.article_id, COALESCE(a.title, f.title_snapshot)
    ORDER BY favoriteCount DESC, f.article_id DESC
    LIMIT 5
</select>

    GROUP BY DATE_FORMAT(f.created_at, '%Y-%m-%d')

  { key: 'pending', label: '待审核', tone: 'warning' },

const PROGRESS_LABELS = ['浅读', '阅读中', '接近读完', '已读完']

function list(value) {
  return Array.isArray(value) ? value : []
}

function number(value) {
  const parsed = Number(value)
  return Number.isFinite(parsed) && parsed >= 0 ? parsed : 0
}

function percent(count, total) {
  if (!total) return 0
  return Math.round((count / total) * 100)
}

function metric(metrics, key, fallback = 0) {
  if (Object.prototype.hasOwnProperty.call(metrics, key)) return number(metrics[key])
  return fallback
}

function normalizeDateBuckets(value) {
  return list(value).map(item => ({
    date: typeof item?.date === 'string' ? item.date : '',
    count: number(item?.count)
  }))
}

function normalizeProgressBuckets(value) {
  const items = list(value)
  const total = items.reduce((sum, item) => sum + number(item?.count), 0)
  const byLabel = new Map(items.map(item => [item?.label, item]))
  return PROGRESS_LABELS
    .map(label => {
      const item = byLabel.get(label) || {}
      const count = number(item.count)
      return { label, count, percent: percent(count, total) }
    })
    .filter(item => item.count > 0 || total > 0)
}

function normalizeTopArticles(value) {
  return list(value)
    .slice(0, 5)
    .map(item => ({
      id: item?.id ?? null,
      title: typeof item?.title === 'string' ? item.title : '未命名文章',
      viewCount: number(item?.viewCount)

    const count = metric(metricsSource, `${item.key}Articles`, number(backendItem.count))
    return {
      ...item,
      count,

    }
  })

function formatNumber(value) {

function formatCompact(value) {
  const parsed = Number(value)
  if (!Number.isFinite(parsed) || parsed <= 0) return '0'
  if (parsed >= 10000) return `${(parsed / 10000).toFixed(1)}w`
  if (parsed >= 1000) return `${(parsed / 1000).toFixed(1)}k`
  return String(parsed)
}

function formatBytes(value) {
  const parsed = Number(value)
  if (!Number.isFinite(parsed) || parsed <= 0) return '0 B'
  const units = ['B', 'KB', 'MB', 'GB', 'TB']
  let size = parsed
  let unit = 0
  while (size >= 1024 && unit < units.length - 1) {
    size /= 1024
    unit += 1
  }
  const digits = size >= 100 ? 0 : size >= 10 ? 1 : 2
  return `${size.toFixed(digits)} ${units[unit]}`
}
